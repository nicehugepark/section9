"""실브라우저에서 진짜 조합(IME) 이벤트로 확인 — REQ-20260911-001-2n7g.

CDP Input.imeSetComposition 으로 실제 조합을 걸고, 조합이 살아 있는 동안 Enter 를
보낸 뒤 확정(insertText)한다. 블링크가 keydown 에 isComposing 을 붙이는 것도,
확정 글자가 값에 앉는 순서도 전부 브라우저가 하는 그대로다.
"""
import json
import os
import re
import subprocess
import sys
import time
import urllib.request

sys.path.insert(0, "/tmp")
from cdpreal import WS, chrome_path                      # noqa: E402

ROOT = "/home/sjpark/section9/state/worktrees/w-911-001-2n7g"
SRC = open(os.environ.get("IME_SRC", "/tmp/alt-input.js"), encoding="utf-8").read()


def grab(name):
    m = re.search(r"\nfunction %s\(T(?:, e)?\)\{\n(?:.*?)\n\}\n" % name, SRC, re.S)
    assert m, name
    return m.group(0)


m0 = re.search(r"const IME_SEND_MS = (\d+);", SRC)
IME_MS = m0.group(1) if m0 else "300"

PAGE = """<!doctype html><html lang="ko"><meta charset="utf-8"><body>
<textarea id="chat-in"></textarea>
<div id="cc-root"></div><div id="cc-pal"></div><div id="cc-att"></div>
<div id="cc-clip"></div><input id="cc-file"><div id="cc-paste"></div>
<script>
const IME_SEND_MS = %s;
%s
%s
function $(s){ return document.querySelector(s); }
let sent = [], picks = 0;
const ta = $("#chat-in");
function sendChat(){ sent.push(ta.value); ta.value = ""; }
function termPalPick(){ picks++; }
function termPalRender(){} function termPalClose(){} function termPalUpdate(){}
function termInputSync(){} function termPasteHint(){} function termAttRender(){}
function termInterrupt(){} function termUpload(){} function termModelChange(){}
function termSessionPick(){} function evEl(x){ return null; }
function pasteFoldNeeded(){ return false; }
function termPasteFindChip(){ return null; } function termPasteFold(){}
const T = {pal:{open:false, idx:0, items:[]}, hist:[], histIdx:null, atts:[],
           composing:false, imeSend:false, imeSendAt:0};
let TERM = T;
window.__log = [];
["compositionstart","compositionend","keydown","input"].forEach(t =>
  ta.addEventListener(t, e => window.__log.push(
    t + (t === "keydown" ? "(" + e.key + ",isComposing=" + e.isComposing + ")" : "")
      + "[" + ta.value + "]"), true));
termBindInput(T);
ta.focus();
</script></body></html>""" % (IME_MS, grab("termBindInput"), grab("termKeydown"))

path = "/tmp/s9-ime-verify.html"
open(path, "w", encoding="utf-8").write(PAGE)

chrome = chrome_path(pixel_contract=False)
assert chrome, "Chrome 을 못 찾았다"
prof = "/tmp/s9-ime-prof"
proc = subprocess.Popen(
    [chrome, "--headless=new", "--no-sandbox", "--disable-gpu",
     "--user-data-dir=" + prof, "--remote-debugging-port=9341",
     "--disable-background-networking", "file://" + path],
    stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
try:
    page = None
    for _ in range(80):
        try:
            tabs = json.loads(urllib.request.urlopen(
                "http://127.0.0.1:9341/json", timeout=2).read().decode())
            page = next((t for t in tabs if t.get("type") == "page"
                         and t.get("webSocketDebuggerUrl")), None)
            if page:
                break
        except Exception:
            pass
        time.sleep(0.25)
    assert page, "CDP 연결 실패"
    ws = WS(page["webSocketDebuggerUrl"])
    time.sleep(0.5)
    ws.eval('$("#chat-in").focus(), 1')

    # 조합 시작 — 「스스로 판단해」가 조합 중인 상태(마지막 글자가 미확정)
    ws.call("Input.imeSetComposition", text="스스로 판단해",
            selectionStart=7, selectionEnd=7)
    time.sleep(0.2)
    # 조합이 살아 있는 채로 Enter
    ws.call("Input.dispatchKeyEvent", type="rawKeyDown", windowsVirtualKeyCode=13,
            nativeVirtualKeyCode=13, key="Enter", code="Enter")
    ws.call("Input.dispatchKeyEvent", type="keyDown", windowsVirtualKeyCode=13,
            nativeVirtualKeyCode=13, key="Enter", code="Enter", text="\r")
    time.sleep(0.05)
    # 입력기가 그 Enter 로 조합을 확정한다
    ws.call("Input.insertText", text="스스로 판단해")
    time.sleep(0.5)
    r = ws.eval('JSON.stringify({sent, left: $("#chat-in").value, log: window.__log})')
    print(r)
finally:
    proc.terminate()
