## Original

REQ-20260910-053 을 맥에서 이어받아 실측한 결과. bin/s9.py cmd_shot 의 chrome 갈래는 후보를 (nt) ProgramFiles 계열 · /mnt/c/... · PATH 의 google-chrome/chromium/chromium-browser 에서만 찾는다. 맥에는 그 셋이 다 없어 「브라우저를 찾지 못했습니다. WSL이면 …」로 exit 3 이고, --engine chromium/webkit 은 _shot_webkit 이 「이 환경(macOS)에는 WebKit 경로가 없다」로 exit 3 이다. 규약(PROTOCOL 19조: 맥에서 Chrome 과 Safari 둘 다 본다)이 요구하는 검증을 하네스 도구가 맥에서 전혀 못 한다. 제안: cmd_shot 후보에 sys.platform == "darwin" 갈래를 더한다 — "/Applications/Google Chrome.app/Contents/MacOS/Google Chrome" · "/Applications/Chromium.app/Contents/MacOS/Chromium" · "/Applications/Microsoft Edge.app/Contents/MacOS/Microsoft Edge" 와 ~/Applications 아래 같은 이름(os.path.expanduser). 종전 --headless --screenshot 경로가 그대로 통하므로 후보만 더하면 된다(REQ-20260910-015 의 chrome_path 맥 갈래와 같은 목록으로 한 곳에 모아 쓴다). Safari 는 safaridriver(-p 4445, W3C /session · /screenshot) 갈래를 --engine safari 로 두는 것이 다음 단계이나 이 요청의 범위는 Chrome 후보까지다. 시험: 맥 후보 경로가 os.path.exists 로 잡히면 candidates 에 드는지(모의 파일), 후보가 없을 때의 안내 문구에 「WSL이면」 대신 판별 문구가 서는지.

**1. bin/s9.py — `def cmd_shot`(현 26191행) 바로 위에 추가**

주석으로 적을 배경(코드에 옮겨 적어라): 맥의 Chrome 계열 후보 (REQ-20260911-002). 종전 후보는 윈도우 디스크(/mnt/c)·리눅스 PATH 이름뿐이라 맥에서는 후보가 0개였다 — 규약(PROTOCOL 19조)이 요구하는 맥 Chrome 검증을 하네스 도구가 못 했다. tests/cdpreal.py 의 MAC_CANDIDATES 는 이 목록(/Applications 기준)의 부분집합이어야 한다 — test_shot_mac 이 지킨다.

```python
SHOT_MAC_APPS = (
    "Google Chrome.app/Contents/MacOS/Google Chrome",
    "Chromium.app/Contents/MacOS/Chromium",
    "Microsoft Edge.app/Contents/MacOS/Microsoft Edge",
)


def _shot_mac_candidates():
    """맥에서 찾을 브라우저 실행 파일 — /Applications 다음 ~/Applications 순."""
    out = []
    for base in ("/Applications", os.path.expanduser("~/Applications")):
        for rel in SHOT_MAC_APPS:
            p = os.path.join(base, rel)
            if os.path.exists(p):
                out.append(p)
    return out


def _shot_manual_hint(url):
    """브라우저를 못 찾았거나 캡처가 실패했을 때 사람이 손으로 열어 볼 명령 —
    판별은 os_platform() 이 한다(종전에는 어느 판이든 「WSL이면 …」이라고 했다)."""
    plat = os_platform()
    if plat == "macOS":
        return ("/Applications 에 Google Chrome(또는 Chromium·Microsoft Edge)이 있는지 "
                f"확인하세요. 수동 확인: \`open {url}\`")
    if plat == "WSL":
        return f"Windows에 Chrome/Edge 설치를 확인하세요. 수동 확인: \`cmd.exe /c start {url}\`"
    if plat == "Windows":
        return f"Chrome/Edge 설치를 확인하세요. 수동 확인: \`start {url}\`"
    return f"PATH 에 google-chrome·chromium 이 있는지 확인하세요. 수동 확인: \`xdg-open {url}\`"
```

```python
"""s9 shot 맥 후보 (REQ-20260911-002) — 맥에서 후보가 0개이던 결함의 회귀 시험.

2026-09-11 실측: 맥에서 \`s9 shot --engine chrome\` 이 /Applications 의 Chrome 을
두고도 「브라우저를 찾지 못했습니다. WSL이면 …」로 끝났다. 후보 목록에 윈도우
디스크와 리눅스 PATH 이름밖에 없었다.

실행: python3 tests/ shot_mac
"""
import importlib.machinery
import importlib.util
import io
import os
import subprocess
import sys
import tempfile
import unittest
from unittest import mock

HERE = os.path.dirname(os.path.abspath(__file__))
S9 = os.path.join(HERE, "..", "bin", "s9")
sys.path.insert(0, HERE)
import cdpreal  # noqa: E402

TMP = tempfile.mkdtemp(prefix="s9-shotmac-")
_prev = {k: os.environ.get(k) for k in ("S9_ROOT", "S9_MACHINE")}
os.environ["S9_ROOT"] = TMP
os.environ["S9_MACHINE"] = "testbox"
try:
    spec = importlib.util.spec_from_loader(
        "s9_mod_shotmac", importlib.machinery.SourceFileLoader("s9_mod_shotmac", S9))
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
finally:
    for k, v in _prev.items():
        if v is None:
            os.environ.pop(k, None)
        else:
            os.environ[k] = v

CHROME = "/Applications/Google Chrome.app/Contents/MacOS/Google Chrome"
HOME_EDGE = os.path.expanduser(
    "~/Applications/Microsoft Edge.app/Contents/MacOS/Microsoft Edge")
URL = "http://127.0.0.1:9909/"


class Args:
    def __init__(self, **kv):
        self.url = URL
        self.out = os.path.join(TMP, "shot.png")
        self.size = "1440,900"
        self.wait = "3000"
        self.__dict__.update(kv)


def run_shot(present):
    """맥으로 가장하고 cmd_shot 을 격리 실행 — present 에 든 경로만 존재한다.

    반환: (subprocess.run 호출 인자 목록, SystemExit 코드 or None, stderr)"""
    calls = []
    args = Args()

    def fake_run(argv, *a, **kw):
        calls.append(argv)
        with open(args.out, "wb") as f:
            f.write(b"\\x89PNG")
        return mock.Mock(returncode=0, stdout="", stderr="")

    real_exists = os.path.exists

    def exists(p):
        p = str(p)
        if p in present:
            return True
        if "/Applications/" in p or p.startswith("/mnt/c/"):
            return False
        return real_exists(p)

    err, code = io.StringIO(), None
    with mock.patch.object(mod, "os_platform", lambda: "macOS"), \\
         mock.patch.object(subprocess, "run", fake_run), \\
         mock.patch.object(mod.os.path, "exists", exists), \\
         mock.patch.object(mod.shutil, "which", lambda n: None), \\
         mock.patch.object(mod, "_reclaim_shot_procs", lambda m, win=False: 0), \\
         mock.patch.object(mod, "_sweep_before_shot", lambda: {"alive": 0}), \\
         mock.patch.object(sys, "stderr", err):
        try:
            mod.cmd_shot(args)
        except SystemExit as e:
            code = e.code
    try:
        os.remove(args.out)
    except OSError:
        pass
    return calls, code, err.getvalue()


class TestShotMac(unittest.TestCase):
    def test_mac_candidates_found(self):
        """정상: /Applications 의 Chrome 을 찾아 headless 캡처 명령을 만든다."""
        calls, code, err = run_shot({CHROME})
        self.assertIsNone(code, err)
        self.assertEqual(calls[0][0], CHROME, calls)
        self.assertIn("--headless=new", calls[0])
        self.assertTrue(any(a.startswith("--screenshot=") for a in calls[0]), calls[0])
        calls, code, _ = run_shot({HOME_EDGE})       # ~/Applications 만 있어도 잡힌다
        self.assertIsNone(code)
        self.assertEqual(calls[0][0], HOME_EDGE)
        calls, _c, _e = run_shot({CHROME, HOME_EDGE})  # 둘 다면 /Applications 먼저
        self.assertEqual(calls[0][0], CHROME)

    def test_mac_none_hint(self):
        """실패: 후보가 없으면 판별 문구와 맥의 여는 명령으로 비-0."""
        calls, code, err = run_shot(set())
        self.assertEqual(code, 3, err)
        self.assertEqual(calls, [])
        self.assertIn("macOS", err)
        self.assertIn(f"open {URL}", err)
        self.assertNotIn("WSL이면", err)

    def test_list_matches_cdpreal(self):
        """회귀: 시험 헬퍼(cdpreal)의 맥 목록과 도구의 목록이 갈리지 않는다."""
        ours = [os.path.join("/Applications", r) for r in mod.SHOT_MAC_APPS]
        for p in cdpreal.MAC_CANDIDATES:
            self.assertIn(p, ours)


if __name__ == "__main__":
    unittest.main()
```
